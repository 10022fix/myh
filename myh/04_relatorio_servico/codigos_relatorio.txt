import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { Observable } from 'rxjs';
import { Aluno } from './aluno';

export interface NovoAluno {
  ra: string;
  nome: string;
  turma: string;
  sala: string;
  periodo: string;
}

@Injectable({
  providedIn: 'root'
})
export class AlunoService {
  private readonly urlRelatorio = 'https://professorbarth.com.br/api-alunos/alunos/listarAlunos.php';
  private readonly urlCadastro = 'https://professorbarth.com.br/api-alunos/alunos/criarAluno.php';

  constructor(private readonly http: HttpClient) {}

  listar(): Observable<Aluno[]> {
    return this.http.get<Aluno[]>(this.urlRelatorio);
  }

  cadastrar(aluno: NovoAluno): Observable<unknown> {
    return this.http.post(this.urlCadastro, aluno);
  }
}

import { Component, OnInit } from '@angular/core';
import { Aluno } from '../aluno';
import { AlunoService } from '../aluno.service';

@Component({
  selector: 'app-relatorio',
  standalone: true,
  templateUrl: './relatorio.component.html',
  styleUrl: './relatorio.component.css'
})
export class RelatorioComponent implements OnInit {
  alunos: Aluno[] = [];
  carregando = true;
  erro = false;

  constructor(private readonly alunoService: AlunoService) {}

  ngOnInit(): void {
    this.alunoService.listar().subscribe({
      next: (alunos) => {
        this.alunos = alunos;
        this.carregando = false;
      },
      error: () => {
        this.erro = true;
        this.carregando = false;
      }
    });
  }
}
